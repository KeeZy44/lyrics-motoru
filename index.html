async function generateVideoWithBackground() { const audioInput = document.getElementById('audioFile').files[0]; const lyrics = document.getElementById('lyricsText').value.trim(); const watermark = document.getElementById('watermarkText').value.trim(); if (!audioInput || !lyrics) { alert("Lütfen ses dosyasını seçin ve sözleri girin!"); return; } const btn = document.getElementById('processBtn'); const box = document.getElementById('status-box'); const video = document.getElementById('outputVideo'); btn.disabled = true; box.style.display = 'block'; // Önce eski durumları temizle ['s1', 's2', 's3'].forEach(id => { const el = document.getElementById(id); el.classList.remove('active', 'done'); }); const step = (id, status) => { const el = document.getElementById(id); el.classList.remove('active', 'done'); if (status) el.classList.add(status); }; try { // 1. Supabase bağlantısı ve dosya kontrolü step('s1', 'active'); const { data: files, error: listError } = await supabaseClient.storage .from('manzaralar') .list('', { limit: 100, offset: 0, sortBy: { column: 'name', order: 'asc' } }); if (listError) { throw new Error( 'Supabase bağlantı hatası: ' + listError.message ); } if (!files || files.length === 0) { throw new Error( 'manzaralar klasöründe hiç dosya bulunamadı!' ); } step('s1', 'done'); // 2. Gerçek video dosyasını bul step('s2', 'active'); const validFiles = files.filter(file => file.name && /\.(mp4|webm|mov|m4v)$/i.test(file.name) ); if (validFiles.length === 0) { throw new Error( 'Supabase manzaralar havuzunda MP4 veya WEBM video bulunamadı!' ); } const randomFile = validFiles[Math.floor(Math.random() * validFiles.length)]; // Dosya için erişilebilir imzalı bağlantı oluştur const { data: signed, error: signedError } = await supabaseClient.storage .from('manzaralar') .createSignedUrl(randomFile.name, 3600); if (signedError || !signed?.signedUrl) { throw new Error( 'Video bağlantısı oluşturulamadı: ' + (signedError?.message || 'Bilinmeyen hata') ); } const videoUrl = signed.signedUrl; // 3. Video gerçekten yüklenene kadar bekle video.pause(); video.removeAttribute('src'); video.load(); await new Promise((resolve, reject) => { const timeout = setTimeout(() => { cleanup(); reject(new Error( 'Video 15 saniye içinde yüklenemedi. Dosya veya bağlantıyı kontrol edin.' )); }, 15000); const cleanup = () => { clearTimeout(timeout); video.removeEventListener('loadedmetadata', onLoaded); video.removeEventListener('error', onError); }; const onLoaded = () => { cleanup(); resolve(); }; const onError = () => { const code = video.error?.code; cleanup(); reject(new Error( 'Video oynatılamadı. Video hata kodu: ' + code + '. Dosya formatını ve Supabase erişimini kontrol edin.' )); }; video.addEventListener('loadedmetadata', onLoaded); video.addEventListener('error', onError); video.src = videoUrl; video.load(); }); // Önizleme göster document.getElementById('preview-container').style.display = 'block'; const downloadBtn = document.getElementById('downloadBtn'); downloadBtn.href = videoUrl; downloadBtn.download = 'turan-medya-video.mp4'; step('s2', 'done'); step('s3', 'done'); alert( 'Arka plan videosu başarıyla yüklendi!\n\n' + 'Dosya: ' + randomFile.name + '\nSüre: ' + video.duration.toFixed(2) + ' saniye' ); } catch (err) { console.error('VIDEO HATASI:', err); alert('HATA: ' + err.message); step('s3', ''); } finally { btn.disabled = false; } }
